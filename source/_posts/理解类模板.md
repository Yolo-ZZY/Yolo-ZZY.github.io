# 类模板

以写一个栈为例，以前只能int型，用模板后就灵活了
```c++

template<typename T>
class SeqStack  //模板名称+类型参数列表=类名称，此处SeqStcak只是模板
{
    public:
    //构造和析构 不用加类型参数列表，别的要加上类型参数列表
    SeqStack<T>(int size=10)   //构造函数列表，下面三行 为初始化，即_pstack = new T[_size];  _size = size;_top = 0; 
        ：_pstack(new T[size])
        ,_top(0)
        ,_size(size)
    {}
    ~SeqStack<T>()
    {
        delete []_pstack;   //  释放整个数组，直接使用 delete _pstack释放那一个单元
        _pstack=nullptr;
    }
    SeqStack<T>(const SeqStack<T>&stack)   // 重写拷贝构造与赋值
        :_top(stack._top)
        ,_size(stack._size)
    {
        for(int i=0;i<_top;i++)
        {
            _pstack[i]=stack._pstack[i];
        }
    }
    SeqStack<T>&operator=(const SeqStack<T>&stack)  //赋值运算符的重写函数
    {
        if(this==&stack)    //防止自溢出
        {
            return *this;
        }
        delete []_pstack;
        _pstack=new T[stack._size];
        _top=stack._top;
        _size=stack._size;
        for(int i=0;i<_top;i++)
        {
            _pstack[i]=stack._pstack[i];
        }
        return *this;
    }
    void push(const T &val);    //只声明，类外实现

    void pop()
    {
        if(empty())
        {
            return;
        }
        _top--;
    }
    T top()const    //读栈顶元素，对于只读，写成const,返回类型为const T
    {
        if (empty())
        {
            throw "stack is empty"; //抛出异常，也是一种返回值且不用和返回值类型一致
        }
        return _pstack[_top-1];
    }

    bool full()const
    {
        return _top==_size;
    }

    bool empty()const
    {
        return _top==0;
    }


    private:
    T *_pstack;     // 这里以前用int，现在用T 模板
    int _top;
    int _size;
    void expand()   //顺序栈底层数组按2倍的方式进行扩容
    {
        T *ptmp=new T[_size*2];
        for(int i=0;i<_top;i++)
        {
            ptmp[i]=_pstack[i];
        }
        delete []_pstack;
        _pstack=ptmp;
        _size*=2;
    }
};
//类外实现成员方法，前面要有类作用域，但是这里已经不认识模板了，一个template只有一个{}能认识
template<typename T>
void SeqStack<T>::push(const T &val)    //类外实现要加类作用域，类=模板+<T>
{
        if(full())
        {
            expand();

        }
        _pstack[_top++]=val;
}   

int main()
{

    SeqStack<int>stack; //这里创建了一个实例，但因为stack是个模板类，所以，里面的成员方法只有下面调用后才出现，因此高效，都按照这样写就行了
    stack.push(1);
    stack.push(2);
    stack.push(3);
    stack.pop();
    stack.pop();


}
```
