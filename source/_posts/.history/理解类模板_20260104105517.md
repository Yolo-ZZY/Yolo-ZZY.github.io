# 类模板
```c++

template<typename T>
class SeqStack  //模板名称+类型参数列表=类名称
{
    public:
    //构造和析构 不用加类型参数列表，别的要加上类型参数列表
    SeqStack(int size=10)   //构造函数
        ：_pstack(new T[size])
        ,_top(0)
        ,_size(size)
    {}
    ~SeqStack()
    {
        delete []_pstack;
        _pstack=nullptr;
    }
    SeqStack(const SeqStack<T>&stack)
        :_top(stack._top)
        ,_size(stack._size)
    {
        for(int i=0;i<_top;i++)
        {
            _pstack[i]=stack._pstack[i];
        }
    }
    SeqStack<T>&operator=(const SeqStack<T>&stack)
    {
        if(this!=&stack)
        {
            return *this;
        }
        delete []_pstack;
        _pstack=new T[stack._size];
        _top=stack._top;
        _size=stack._size;
        for(int i=0;i<_top;i++)
        {
            _pstack[i]=stack._pstack[i];
        }
        return *this;
    }
    void push(const T &val);

    void pop()
    {
        if(empty())
        {
            return;
        }
        _top--;
    }
    T top()const    //对于只读，写成const,返回类型为const T
    {
        if (empty())
        {
            throw "stack is empty"; //抛出异常，不用管返回值
        }
        return _pstack[_top-1];
    }

    bool full()const
    {
        return _top==_size;
    }

    bool empty()const
    {
        return _top==0;
    }


    private:
    T *_pstack;
    int _top;
    int _size;
    void expand()   //顺序栈底层数组按2倍的方式进行扩容
    {
        T *ptmp=new T[_size*2];
        for(int i=0;i<_top;i++)
        {
            ptmp[i]=_pstack[i];
        }
        delete []_pstack;
        _pstack=ptmp;
    }
};
//类外实现成员方法，前面要有类作用域，但是这里已经不认识模板了
template<typename T>
void SeqStack<T>::push(const T &val)
{
        if(full())
        {
            expand();

        }
        _pstack[_top++]=val;
}   

int main()
{

    SeqStack<int>stack;
    stack.push(1);
    stack.push(2);
    stack.push(3);
    stack.pop();
    stack.pop();


}
```
